<script>
// Enter key support
document.getElementById("movie").addEventListener("keypress", function(e) {
    if (e.key === "Enter") searchMovie();
});

function searchMovie() {
    let movie = document.getElementById("movie").value.trim();
    let result = document.getElementById("result");

    if (!movie) {
        result.innerHTML = "<p>⚠️ Please enter a movie name</p>";
        return;
    }

    // Loader
    result.innerHTML = '<div class="loader"></div>';

    fetch(`https://api.tvmaze.com/search/shows?q=${movie}`)
        .then(res => res.json())
        .then(data => {

            if (data.length === 0) {
                result.innerHTML = "<p>❌ No movies found</p>";
                return;
            }

            let output = "";

            data.forEach(item => {
                let show = item.show;

                let image = show.image 
                    ? show.image.medium 
                    : "https://via.placeholder.com/250x180?text=No+Image";

                let rating = show.rating.average 
                    ? `<span class="rating">⭐ ${show.rating.average}</span>` 
                    : `<span class="rating">⭐ N/A</span>`;

                output += `
                    <div class="card" onclick="openDetails('${show.url}')">
                        <img src="${image}">
                        <h3>${show.name}</h3>
                        <p>${show.language}</p>
                        ${rating}
                    </div>
                `;
            });

            result.innerHTML = output;

        })
        .catch(() => {
            result.innerHTML = "<p>⚠️ Error fetching data. Try again.</p>";
        });
}

// Open show details
function openDetails(url) {
    window.open(url, "_blank");
}
