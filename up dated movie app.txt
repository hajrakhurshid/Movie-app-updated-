<script>
function searchMovie() {
    let movie = document.getElementById("movie").value.trim();
    let result = document.getElementById("result");

    if (!movie) {
        result.innerHTML = "⚠️ Enter a movie name";
        return;
    }

    result.innerHTML = "Loading... ⏳";

    // 🌐 API call (Fetch API)
    fetch(`https://api.tvmaze.com/search/shows?q=${movie}`)
        .then(res => res.json())
        .then(data => {

            let output = "";

            if (data.length > 0) {

                // 🔁 Dynamic Content Generation (Day 18)
                data.forEach(item => {
                    let show = item.show;

                    output += `
                        <div class="card">
                            <img src="${show.image ? show.image.medium : ''}">
                            <h3>${show.name}</h3>
                            <p><b>Language:</b> ${show.language}</p>
                            <p><b>Rating:</b> ${show.rating.average || "N/A"}</p>
                        </div>
                    `;
                });

                result.innerHTML = output;

            } else {
                result.innerHTML = "❌ No movies found";
            }

        })
        .catch(err => {
            result.innerHTML = "⚠️ Error fetching data";
        });
}
</script>
